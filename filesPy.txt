_telemetry_lock = threading.Lock()
_last_telemetry_signature = ""


def write_telemetry_snapshot(state: str, active_tool: str, last_action: str) -> None:
    """Guarda el estado instantáneo del agente en disco solo cuando cambia."""
    global _last_telemetry_signature
    clean_action = last_action[:150]
    sig = f"{state}|{active_tool}|{clean_action}"

    with _telemetry_lock:
        if sig == _last_telemetry_signature:
            return
        _last_telemetry_signature = sig
        payload: Dict[str, Any] = {
            "status": state,
            "tool": active_tool,
            "last_action": clean_action,
            "updated_at": datetime.now().strftime("%H:%M:%S"),
        }
        try:
            TELEMETRY_FILE.write_text(
                json.dumps(payload, ensure_ascii=False, indent=2), encoding="utf-8"
            )
        except Exception:
            pass

class PollingLogFilter(logging.Filter):
    """Oculta únicamente el polling de alta frecuencia de Rainmeter y el HUD."""
    def filter(self, record: logging.LogRecord) -> bool:
        msg = record.getMessage()
        return "GET /telemetry" not in msg and "GET /api/state" not in msg

class ArgonHermesOrchestrator:
    def __init__(self) -> None:
        self.state = "IDLE"
        self.active_tool = "NONE"
        self.last_action = "Núcleo Argon 4 // Hermes inicializado con Google Gemini."
        self.handsfree = False
        self.ui_logs: List[Dict[str, str]] = []
        self._state_lock = threading.RLock()

        # Pool de conexiones HTTP persistentes para baja latencia
        self.http = requests.Session()
        adapter = HTTPAdapter(
            pool_connections=10,
            pool_maxsize=20,
            max_retries=Retry(total=2, backoff_factor=0.2)
        )
        self.http.mount("http://", adapter)
        self.http.mount("https://", adapter)

        self.voice = VoiceEngine(on_state_change=self._on_state_change)
        self.tasks = SystemTasks()
        self.whatsapp = WhatsAppBridge()
        self.voip = VoIPCaller()
        self.productivity = ProductivityTools()
        self.skills = AdvancedSkills()
        self.notion = NotionBridge()

        # Exportar skin de Rainmeter automáticamente al iniciar
        self.notion.export_rainmeter_skin()

        self.active_model: str = Config.OLLAMA_MODEL
        self.gemini_online: bool = bool(Config.GEMINI_API_KEY)
        self.ollama_online: bool = self._detect_ollama()

        self.system_prompt = (
            f"Eres {Config.AGENT_NAME} (Edición Hermes), un sistema agéntico multimodal avanzado estilo J.A.R.V.I.S. "
            "impulsado por Google Gemini y ejecutándose en Windows 11 junto a Franco. Tu voz es grave, serena y futurista. "
            "Cuentas con memoria persistente SQLite, agenda sincronizada con Notion, triaje de correo y WhatsApp. "
            "Responde siempre en español con precisión ejecutiva en 1 a 3 oraciones claras."
        )
        self.chat_history: List[Dict[str, str]] = [{"role": "system", "content": self.system_prompt}]
        past_turns = self.skills.load_recent_conversation(limit=8)
        if past_turns:
            self.chat_history.extend(past_turns)

        self.router = AgentToolRouter(self)
        write_telemetry_snapshot(self.state, self.active_tool, self.last_action)

    def _on_state_change(self, new_state: str) -> None:
        with self._state_lock:
            self.state = new_state
            write_telemetry_snapshot(self.state, self.active_tool, self.last_action)

    def _set_tool_telemetry(self, tool_name: str, description: str) -> None:
        with self._state_lock:
            self.active_tool = tool_name
            self.last_action = description
            self.state = "TOOL_EXEC" if tool_name != "NONE" else "IDLE"
            write_telemetry_snapshot(self.state, self.active_tool, self.last_action)

    def _detect_ollama(self) -> bool:
        try:
            resp = self.http.get(f"{Config.OLLAMA_HOST}/api/tags", timeout=3)
            if resp.status_code == 200:
                models = [m.get("name") for m in resp.json().get("models", []) if m.get("name")]
                if models:
                    self.active_model = Config.OLLAMA_MODEL if Config.OLLAMA_MODEL in models else models[0]
                    return True
        except requests.RequestException:
            pass
        return False

    def add_log(self, sender: str, text: str, tag: str = "agent") -> None:
        with self._state_lock:
            self.ui_logs.append({"sender": sender, "text": text, "tag": tag})
            if len(self.ui_logs) > 60:
                self.ui_logs = self.ui_logs[-60:]

    def respond(self, text: str) -> None:
        with self._state_lock:
            self.last_action = text
            self.active_tool = "VOICE_SYNTH"
            write_telemetry_snapshot("SPEAKING", self.active_tool, self.last_action)
        self.add_log(Config.AGENT_NAME.upper(), text, "agent")
        self.skills.log_conversation_turn("assistant", text)
        self.voice.speak(text)
        with self._state_lock:
            self.active_tool = "NONE"
            self.state = "IDLE"
            write_telemetry_snapshot("IDLE", self.active_tool, self.last_action)

    def ask_brain(self, user_prompt: str, extra_context: str = "") -> str:
        """Consulta a Google Gemini como motor principal y usa Ollama como respaldo local."""
        memories = self.skills.recall_memories(limit=10)
        mem_block = ""
        if memories:
            facts = "; ".join([f"[{m['category']}] {m['note']}" for m in memories])
            mem_block = f"\n[MEMORIA PERMANENTE SOBRE FRANCO: {facts}]"

        full_user_msg = f"{user_prompt}{mem_block}"
        if extra_context:
            full_user_msg += f"\n{extra_context}"

        with self._state_lock:
            self.chat_history.append({"role": "user", "content": full_user_msg})
            if len(self.chat_history) > 15:
                self.chat_history = [self.chat_history[0]] + self.chat_history[-14:]
            history_snapshot = list(self.chat_history)

        # 1. Motor Principal: Google Gemini API
        if self.gemini_online:
            try:
                self._set_tool_telemetry("GOOGLE_GEMINI", f"Razonando con {Config.GEMINI_MODEL}...")
                url = (
                    f"https://generativelanguage.googleapis.com/v1beta/models/"
                    f"{Config.GEMINI_MODEL}:generateContent?key={Config.GEMINI_API_KEY}"
                )
                gemini_contents = []
                for m in history_snapshot:
                    role = m.get("role")
                    if role == "user":
                        gemini_contents.append({"role": "user", "parts": [{"text": m["content"]}]})
                    elif role == "assistant":
                        gemini_contents.append({"role": "model", "parts": [{"text": m["content"]}]})

                payload = {
                    "systemInstruction": {"parts": [{"text": self.system_prompt}]},
                    "contents": gemini_contents,
                    "generationConfig": {
                        "temperature": 0.4,
                        "maxOutputTokens": 450,
                    },
                }
                resp = self.http.post(url, json=payload, timeout=25)
                if resp.status_code == 200:
                    candidates = resp.json().get("candidates", [])
                    if candidates:
                        parts = candidates[0].get("content", {}).get("parts", [])
                        reply = "".join(p.get("text", "") for p in parts if "text" in p).strip()
                        if reply:
                            with self._state_lock:
                                self.chat_history.append({"role": "assistant", "content": reply})
                            return reply
                else:
                    logger.warning(f"Gemini API HTTP {resp.status_code}: {resp.text[:160]}")
            except Exception as exc:
                logger.warning(f"Fallo en Google Gemini, conmutando a Ollama local: {exc}")
            finally:
                with self._state_lock:
                    self.state = "IDLE"

        # 2. Motor de Respaldo Offline: Ollama Local
        if not self.ollama_online:
            self.ollama_online = self._detect_ollama()
            if not self.ollama_online:
                return "Configura tu GEMINI_API_KEY en el archivo .env o inicia el servidor local Ollama."

        try:
            self._set_tool_telemetry("OLLAMA_LLM", f"Inferencia en {self.active_model}...")
            payload = {
                "model": self.active_model,
                "messages": history_snapshot,
                "stream": False
            }
            resp = self.http.post(
                f"{Config.OLLAMA_HOST}/api/chat",
                json=payload,
                timeout=Config.OLLAMA_TIMEOUT
            )
            if resp.status_code == 200:
                reply = resp.json().get("message", {}).get("content", "").strip()
                if reply:
                    with self._state_lock:
                        self.chat_history.append({"role": "assistant", "content": reply})
                    return reply
        except Exception as exc:
            logger.error(f"Error consultando Ollama: {exc}")
        finally:
            with self._state_lock:
                self.state = "IDLE"
        return "Tuve una breve latencia en el motor neuronal. ¿Me lo repites?"

    def get_unified_calendar(self) -> List[Dict[str, Any]]:
        """Combina los pendientes locales con las tareas de Notion API si están activas."""
        local_events = self.productivity.list_calendar_events()
        notion_events = self.notion.fetch_tasks(limit=4)
        return notion_events + local_events

    def process_command(self, user_text: str) -> None:
        text = user_text.lower().strip()
        self.skills.log_conversation_turn("user", user_text)
        self.skills.auto_extract_memory(user_text)

        # Si el usuario encadena acciones múltiples ("y", "luego", "además"), usar el ToolRouter paralelo
        if any(conn in text for conn in [" y ", " luego ", " además ", " ademas "]):
            routed_compound = self.router.route_and_execute(user_text)
            if routed_compound:
                self.respond(routed_compound)
                return

        # 1. MEMORIA PERSISTENTE
        if any(k in text for k in ["recuerda que", "memoriza que", "guarda en tu memoria", "no olvides que"]):
            self._set_tool_telemetry("SQLITE_MEMORY", "Guardando recuerdo permanente")
            cleaned = re.sub(
                r"^(recuerda que|memoriza que|guarda en tu memoria|no olvides que)\s*:?\s*",
                "",
                user_text,
                flags=re.IGNORECASE
            ).strip()
            res = self.skills.save_memory(cleaned, category="Clave")
            self.respond(f"Registrado en mi núcleo de memoria permanente: {cleaned}." if res["ok"] else res["error"])
            return

        if any(k in text for k in ["qué recuerdas", "que recuerdas", "qué sabes de mí", "que sabes de mi"]):
            self._set_tool_telemetry("SQLITE_MEMORY", "Consultando recuerdos permanentes")
            mems = self.skills.recall_memories(limit=6)
            if not mems:
                self.respond("Aún estamos construyendo mi base de recuerdos permanentes contigo.")
            else:
                notes = ". ".join([m["note"] for m in mems])
                self.respond(f"Tengo presente en mi memoria permanente lo siguiente: {notes}.")
            return

        # 2. ACCESO DIRECTO Y AUTO-ARRANQUE WINDOWS 11
        if any(k in text for k in ["acceso directo", "arrancar con windows", "inicio automático", "inicio automatico"]):
            self._set_tool_telemetry("WIN11_DEPLOY", "Configurando acceso directo / inicio")
            enable_boot = "arrancar" in text or "inicio" in text
            res = self.skills.setup_windows_integration(enable_startup=enable_boot)
            self.respond(res["message"] if res["ok"] else res["error"])
            return

        # 3. INTELIGENCIA DE PORTAPAPELES
        if any(k in text for k in ["portapapeles", "lo que copié", "lo que copie"]):
            self._set_tool_telemetry("CLIPBOARD_AI", "Leyendo portapapeles del sistema")
            res = self.skills.read_clipboard()
            if not res["ok"]:
                self.respond(res["error"])
                return
            self.respond("Inspeccionando el contenido de tu portapapeles.")
            reply = self.ask_brain(
                "Analiza lo que el usuario tiene en su portapapeles y dale una respuesta directa y útil:",
                extra_context=f"[PORTAPAPELES]:\n{res['content']}"
            )
            self.respond(reply)
            return

        # 4. BÚSQUEDA WEB EN TIEMPO REAL
        if any(k in text for k in ["busca en internet", "buscar en internet", "noticias de", "investiga sobre"]):
            self._set_tool_telemetry("WEB_SEARCH", "Consultando resultados web en vivo")
            query = re.sub(
                r"^(busca en internet|buscar en internet|investiga sobre)\s*:?\s*",
                "",
                user_text,
                flags=re.IGNORECASE
            ).strip() or user_text
            self.respond(f"Rastreando la red en tiempo real sobre {query}.")
            web_res = self.skills.search_web(query)
            if web_res["ok"]:
                reply = self.ask_brain(
                    f"Sintetiza en 2 frases estos resultados web actuales sobre '{query}':",
                    extra_context=web_res["context"]
                )
                self.respond(reply)
            else:
                self.respond(web_res["error"])
            return

        # 5. TELEMETRÍA Y TOP PROCESOS
        if any(k in text for k in ["qué procesos", "que procesos", "consumiendo más ram", "consumiendo mas ram"]):
            self._set_tool_telemetry("PSUTIL_PROC", "Escaneando procesos en RAM")
            res = self.skills.get_top_processes()
            self.respond(f"Los procesos más exigentes en memoria son: {res['summary']}." if res["ok"] else res["error"])
            return

        if any(k in text for k in ["estado del sistema", "telemetría", "telemetria"]):
            self._set_tool_telemetry("SYSTEM_STATUS", "Recolectando biometría de hardware")
            res = self.tasks.get_system_status()
            if res["ok"]:
                d = res["data"]
                self.respond(
                    f"Todos los sistemas estables. CPU al {d['cpu_percent']}%, RAM al {d['ram_percent']}% "
                    f"con {d['ram_available_gb']} gigabytes libres y batería al {d['battery_percent']}%."
                )
            return

        # 6. AGENDA HÍBRIDA (LOCAL + NOTION API)
        if any(k in text for k in ["agendar", "agregar pendiente", "nuevo pendiente", "anota en notion"]):
            self._set_tool_telemetry("NOTION_CALENDAR", "Registrando evento en agenda y Notion")
            cleaned = re.sub(
                r"^(agendar|agregar pendiente|nuevo pendiente|anota en notion)\s*:?\s*",
                "",
                user_text,
                flags=re.I
            ).strip()
            if cleaned:
                self.productivity.add_calendar_event(cleaned)
                notion_res = self.notion.create_task(cleaned)
                if notion_res.get("ok"):
                    self.respond(f"Evento sincronizado en tu agenda local y en tu base de datos de Notion: {cleaned}.")
                else:
                    self.respond(f"Agendado en tu panel táctico local: {cleaned}.")
                return

        if any(k in text for k in ["calendario", "pendientes", "qué tengo hoy", "que tengo hoy", "notion"]):
            self._set_tool_telemetry("NOTION_CALENDAR", "Consultando agenda local y Notion")
            evs = self.get_unified_calendar()
            if not evs:
                self.respond("No tienes pendientes activos en tu agenda ni en Notion.")
            else:
                titles = ", ".join([f"{e['id']}: {e['title']}" for e in evs[:4]])
                self.respond(f"Tus pendientes activos son: {titles}.")
            return

        # 7. CORREOS Y AUDITORÍA DE CÓDIGO
        if any(k in text for k in ["revisar correo", "revisar correos", "mis correos"]):
            self._set_tool_telemetry("IMAP_MAIL", f"Consultando bandeja {Config.EMAIL_USER}")
            res = self.productivity.check_emails()
            self.respond(res["summary"] if res["ok"] else res["error"])
            return

        if any(k in text for k in ["revisar código", "revisar codigo", "analizar código", "analizar codigo"]):
            self._set_tool_telemetry("AST_CODE_AUDIT", "Inspeccionando código fuente")
            file_match = re.search(r"([a-zA-Z0-9_./\\-]+\.(?:py|js|ts|json|txt|md))", user_text)
            target_file = file_match.group(1) if file_match else "main_agent.py"
            res = self.productivity.analyze_code_file(target_file)
            if not res["ok"]:
                self.respond(res["error"])
                return
            reply = self.ask_brain(
                f"Evalúa técnicamente este código ({res['filepath']}, {res['syntax']}):",
                extra_context=res["snippet"]
            )
            self.respond(reply)
            return

        # 8. APLICACIONES, WHATSAPP Y VOIP
        for app_name in ["calculadora", "bloc de notas", "explorador", "terminal"]:
            if f"abrir {app_name}" in text or f"abre la {app_name}" in text:
                self._set_tool_telemetry("OS_SUBPROCESS", f"Abriendo {app_name}")
                res = self.tasks.open_application(app_name)
                self.respond(f"Desplegando {app_name}." if res["ok"] else res["error"])
                return

        if "whatsapp" in text:
            self._set_tool_telemetry("WHATSAPP_BRIDGE", "Despachando mensaje de WhatsApp")
            match = re.search(r"(\+?\d[\d\s\-]{7,}\d).*?(?:diciendo|mensaje|que diga)\s+(.+)", user_text, re.I)
            if match:
                phone, msg = match.group(1).strip(), match.group(2).strip()
            else:
                msg_match = re.search(r"(?:diciendo|mensaje|que diga)\s+(.+)", user_text, re.I)
                phone = Config.DEFAULT_TARGET_PHONE
                msg = msg_match.group(1).strip() if msg_match else ""

            if phone and msg:
                self.respond(f"Transmitiendo WhatsApp a {phone}.")
                res = self.whatsapp.send_message(phone, msg)
                self.respond("Mensaje de WhatsApp entregado." if res["ok"] else res["error"])
            else:
                self.respond("Dime qué mensaje deseas enviar por WhatsApp, por ejemplo: 'Enviar WhatsApp diciendo Hola'.")
            return

        if "sala de voz" in text or "webrtc" in text:
            self._set_tool_telemetry("VOIP_WEBRTC", "Creando sala WebRTC")
            res = self.voip.start_webrtc_room()
            if res["ok"]:
                self.respond("Canal de voz WebRTC abierto.")
            return

        # 9. CONVERSACIÓN PROFUNDA CON MEMORIA PERSISTENTE
        routed_reply = self.router.route_and_execute(user_text)
        if routed_reply is not None:
            self.respond(routed_reply)
            return

        reply = self.ask_brain(user_text)
        self.respond(reply)

@app.route("/")
@app.route("/hud")
def index() -> Response:
    return Response(HUD_HTML, mimetype="text/html")


@app.route("/telemetry")
def rainmeter_telemetry() -> Response:
    """Endpoint ligero JSON consumido por Rainmeter (skins/HUD_Agent.ini)."""
    st = core.tasks.get_system_status()
    d = st["data"] if st["ok"] else {"cpu_percent": 0, "ram_percent": 0, "battery_percent": 100}
    return jsonify({
        "status": core.state,
        "tool": core.active_tool,
        "last_action": core.last_action,
        "updated_at": time.strftime("%H:%M:%S"),
        "cpu": d["cpu_percent"],
        "ram": d["ram_percent"],
        "battery": d["battery_percent"],
    })


@app.route("/api/state")
def api_state() -> Response:
    st = core.tasks.get_system_status()
    telemetry = st["data"] if st["ok"] else {"cpu_percent": 0, "ram_percent": 0, "battery_percent": 100}
    return jsonify({
        "state": core.state,

except ImportError:
    class FastMCP:  # type: ignore[no-redef]
        """Respaldo ligero para evitar errores de importación si 'mcp' no está instalado en .venv."""
        def __init__(self, name: str) -> None:
            self.name = name
            self.tools: List[Callable[..., Any]] = []

        def tool(self) -> Callable[[Callable[..., Any]], Callable[..., Any]]:
            def decorator(func: Callable[..., Any]) -> Callable[..., Any]:
                self.tools.append(func)
                return func
            return decorator

        def run(self) -> None:
            print(f"[{self.name}] Instala el paquete oficial ejecutando: .venv\\Scripts\\pip install mcp")

"""
Ruta: modules/advanced_skills.py
Descripción: Memoria Cognitiva Persistente (hechos + historial entre sesiones en SQLite),
Auto-extracción de recuerdos, Búsqueda Web, Portapapeles y Despliegue en Escritorio/Inicio de Windows.
"""
import os
import re
import sqlite3
import subprocess
import logging
from datetime import datetime
from pathlib import Path
from typing import Dict, Any, List, Optional
import psutil
import pyperclip
from duckduckgo_search import DDGS
from config import BASE_DIR, LOG_DIR

logger = logging.getLogger("AdvancedSkills")
MEMORY_DB = LOG_DIR / "argon_memory.db"


class AdvancedSkills:
    def __init__(self) -> None:
        self._memories_cache: Optional[List[Dict[str, Any]]] = None
        self._init_memory_db()

    @staticmethod
    def _get_conn() -> sqlite3.Connection:
        return sqlite3.connect(MEMORY_DB, timeout=10.0)

    @staticmethod
    def _init_memory_db() -> None:
        """Inicializa tablas para memoria semántica (hechos) y memoria episódica (conversación)."""
        with AdvancedSkills._get_conn() as conn:
            conn.execute("PRAGMA journal_mode=WAL;")
            conn.execute(
                """
                CREATE TABLE IF NOT EXISTS memories (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    category TEXT DEFAULT 'General',
                    note TEXT NOT NULL,
                    created_at TEXT NOT NULL
                )
                """
            )
            cols = [r[1] for r in conn.execute("PRAGMA table_info(memories);")]
            if "category" not in cols:
                conn.execute("ALTER TABLE memories ADD COLUMN category TEXT DEFAULT 'General';")
            conn.execute(
                """
                CREATE TABLE IF NOT EXISTS chat_log (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    role TEXT NOT NULL,
                    content TEXT NOT NULL,
                    timestamp TEXT NOT NULL
                )
                """
            )
            conn.commit()

    # ---------------- 1. MEMORIA PERSISTENTE Y AUTO-APRENDIZAJE ----------------
    def save_memory(self, note: str, category: str = "Usuario") -> Dict[str, Any]:
        clean = note.strip()
        if not clean:
            return {"ok": False, "error": "El contenido a recordar está vacío."}
        try:
            now = datetime.now().strftime("%Y-%m-%d %H:%M")
            with self._get_conn() as conn:
                # Evitar duplicados exactos
                existing = conn.execute("SELECT id FROM memories WHERE lower(note) = ?", (clean.lower(),)).fetchone()
                if not existing:
                    conn.execute(
                        "INSERT INTO memories (category, note, created_at) VALUES (?, ?, ?)",
                        (category, clean, now)
                    )
                    conn.commit()
            self._memories_cache = None
            logger.info(f"Memoria persistente almacenada [{category}]: {clean}")
            return {"ok": True, "note": clean, "category": category}
        except Exception as exc:
            return {"ok": False, "error": str(exc)}

    def delete_memory(self, mem_id: int) -> Dict[str, Any]:
        try:
            with self._get_conn() as conn:
                conn.execute("DELETE FROM memories WHERE id = ?", (mem_id,))
                conn.commit()
            self._memories_cache = None
            return {"ok": True}
        except Exception as exc:
            return {"ok": False, "error": str(exc)}

    def recall_memories(self, limit: int = 10) -> List[Dict[str, Any]]:
        try:
            if self._memories_cache is not None:
                return self._memories_cache[:limit]
            with self._get_conn() as conn:
                cur = conn.execute(
                    "SELECT id, category, note, created_at FROM memories ORDER BY id DESC"
                )
                rows = cur.fetchall()
            self._memories_cache = [{"id": r[0], "category": r[1], "note": r[2], "date": r[3]} for r in rows]
            return self._memories_cache[:limit]
        except Exception as exc:
            logger.error(f"Error leyendo memoria SQLite: {exc}")
            return []

    def auto_extract_memory(self, user_text: str) -> None:
        """
        Detecta automáticamente datos importantes que el usuario menciona en conversación natural
        (gustos, proyectos, objetivos, datos personales) y los guarda sin que tenga que pedirlo.
        """
        patterns = [
            (r"mi proyecto (?:se llama|es|trata sobre)\s+([^.,;\n]+)", "Proyecto"),
            (r"estoy trabajando en\s+([^.,;\n]+)", "Proyecto"),
            (r"prefiero\s+([^.,;\n]+)", "Preferencia"),
            (r"me gusta\s+([^.,;\n]+)", "Preferencia"),
            (r"mi correo (?:es|principal es)\s+([^.,;\n]+)", "Contacto"),
            (r"mi objetivo es\s+([^.,;\n]+)", "Objetivo"),
            (r"tengo que entregar\s+([^.,;\n]+)", "Prioridad"),
        ]
        for pat, cat in patterns:
            match = re.search(pat, user_text, re.IGNORECASE)
            if match and len(match.group(0)) < 140:
                self.save_memory(match.group(0).strip(), category=cat)
                break

    def log_conversation_turn(self, role: str, content: str) -> None:
        """Guarda cada turno de conversación en disco para recordar entre sesiones."""
        try:
            now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            with self._get_conn() as conn:
                conn.execute(
                    "INSERT INTO chat_log (role, content, timestamp) VALUES (?, ?, ?)",
                    (role, content.strip(), now)
                )
                conn.commit()
        except Exception as exc:
            logger.error(f"Error guardando historial de chat: {exc}")

    def load_recent_conversation(self, limit: int = 10) -> List[Dict[str, str]]:
        """Recupera las últimas interacciones de sesiones anteriores al iniciar Argon 4."""
        try:
            with self._get_conn() as conn:
                cur = conn.execute(
                    "SELECT role, content FROM chat_log ORDER BY id DESC LIMIT ?",
                    (limit,)
                )
                rows = cur.fetchall()
            return [{"role": r[0], "content": r[1]} for r in reversed(rows)]
        except Exception:
            return []

    # ---------------- 2. ACCESO DIRECTO Y AUTO-ARRANQUE WINDOWS 11 ----------------
    @staticmethod
    def setup_windows_integration(enable_startup: bool = False) -> Dict[str, Any]:
        """
        Crea un lanzador silencioso (.vbs) + Acceso Directo en el Escritorio de Windows
        y opcionalmente lo registra en el Inicio Automático de Windows (shell:startup).
        """
        try:
            venv_pythonw = BASE_DIR / ".venv" / "Scripts" / "pythonw.exe"
            venv_python = BASE_DIR / ".venv" / "Scripts" / "python.exe"
            python_bin = venv_pythonw if venv_pythonw.exists() else venv_python

            # 1. Crear script .bat y lanzador silencioso .vbs en la raíz del proyecto
            bat_path = BASE_DIR / "Launch_Argon4.bat"
            bat_content = (
                f'@echo off\n'
                f'cd /d "{BASE_DIR}"\n'
                f'"{venv_python}" "{BASE_DIR / "main_agent.py"}"\n'
            )
            bat_path.write_text(bat_content, encoding="utf-8")

            vbs_path = BASE_DIR / "Argon4_Silent_Launcher.vbs"
            vbs_content = (
                f'Set WshShell = CreateObject("WScript.Shell")\n'
                f'WshShell.CurrentDirectory = "{BASE_DIR}"\n'
                f'WshShell.Run """{python_bin}"" ""{BASE_DIR / "main_agent.py"}""", 0, False\n'
            )
            vbs_path.write_text(vbs_content, encoding="utf-8")

            # 2. Crear Acceso Directo (.lnk) en el Escritorio usando PowerShell nativo
            desktop_dir = Path(os.path.expanduser("~")) / "Desktop"
            shortcut_desktop = desktop_dir / "Argon 4 AI.lnk"

            ps_script = (
                f'$WshShell = New-Object -ComObject WScript.Shell; '
                f'$Shortcut = $WshShell.CreateShortcut("{shortcut_desktop}"); '
                f'$Shortcut.TargetPath = "wscript.exe"; '
                f'$Shortcut.Arguments = """{vbs_path}"""; '
                f'$Shortcut.WorkingDirectory = "{BASE_DIR}"; '
                f'$Shortcut.IconLocation = "shell32.dll,13"; '
                f'$Shortcut.Description = "Argon 4 Tactical AI Assistant"; '
                f'$Shortcut.Save()'
            )
            subprocess.run(["powershell", "-NoProfile", "-Command", ps_script], check=True)

            # 3. Configurar o quitar de Inicio Automático de Windows (Startup)
            startup_dir = (
                Path(os.environ.get("APPDATA", ""))
                / "Microsoft" / "Windows" / "Start Menu" / "Programs" / "Startup"
            )
            shortcut_startup = startup_dir / "Argon 4 AI.lnk"

            startup_msg = "Acceso directo creado en tu Escritorio ('Argon 4 AI')."
            if enable_startup and startup_dir.exists():
                ps_startup = ps_script.replace(str(shortcut_desktop), str(shortcut_startup))
                subprocess.run(["powershell", "-NoProfile", "-Command", ps_startup], check=True)
                startup_msg = "Acceso directo creado en el Escritorio y activado en el Inicio Automático de Windows."
            elif not enable_startup and shortcut_startup.exists():
                shortcut_startup.unlink()

            return {"ok": True, "message": startup_msg}
        except Exception as exc:
            logger.error(f"Error configurando acceso directo en Windows: {exc}")
            return {"ok": False, "error": str(exc)}

    # ---------------- 3. BÚSQUEDA WEB, PORTAPAPELES Y PROCESOS ----------------
    @staticmethod
    def search_web(query: str, max_results: int = 3) -> Dict[str, Any]:
        clean_q = query.strip()
        if not clean_q:
            return {"ok": False, "error": "Consulta vacía."}
        try:
            results = []
            with DDGS() as ddgs:
                for r in ddgs.text(clean_q, region="pe-es", max_results=max_results):
                    results.append(f"- {r.get('title', '')}: {r.get('body', '')}")
            if not results:
                return {"ok": False, "error": "Sin resultados web."}
            return {"ok": True, "query": clean_q, "context": "\n".join(results)}
        except Exception as exc:
            return {"ok": False, "error": str(exc)}

    @staticmethod
    def read_clipboard() -> Dict[str, Any]:
        try:
            content = pyperclip.paste()
            if not content or not content.strip():
                return {"ok": False, "error": "Portapapeles vacío."}
            return {"ok": True, "content": content.strip()[:2500]}
        except Exception as exc:
            return {"ok": False, "error": str(exc)}

    @staticmethod
    def get_top_processes(limit: int = 4) -> Dict[str, Any]:
        try:
            procs = []
            for p in psutil.process_iter(["pid", "name", "memory_info"]):
                try:
                    mem_mb = p.info["memory_info"].rss / (1024 * 1024)
                    procs.append({"name": p.info["name"], "mem_mb": round(mem_mb, 1)})
                except (psutil.NoSuchProcess, psutil.AccessDenied):
                    continue
            procs.sort(key=lambda x: x["mem_mb"], reverse=True)
            top = procs[:limit]
            summary = ", ".join([f"{item['name']} ({item['mem_mb']} MB)" for item in top])
            return {"ok": True, "top": top, "summary": summary}
        except Exception as exc:
            return {"ok": False, "error": str(exc)}

"""

Descripción: Conector bidireccional para Notion API y generador automático de la Skin Rainmeter.

import time
import logging
from datetime import datetime
from typing import Dict, Any, List
import requests
from config import Config, SKINS_DIR

logger = logging.getLogger("NotionBridge")


class NotionBridge:
    BASE_URL = "https://api.notion.com/v1"
    NOTION_VERSION = "2022-06-28"

    def __init__(self) -> None:
        self.enabled = bool(Config.NOTION_API_KEY and Config.NOTION_DATABASE_ID)
        self.headers = {
            "Authorization": f"Bearer {Config.NOTION_API_KEY}",
            "Notion-Version": self.NOTION_VERSION,
            "Content-Type": "application/json",
        }
        self._cached_tasks: List[Dict[str, str]] = []
        self._last_fetch_ts: float = 0.0
        self._session = requests.Session()

    def fetch_tasks(self, limit: int = 5) -> List[Dict[str, str]]:
        """Obtiene tareas desde Notion si está configurado en .env."""
        if not self.enabled:
            return []
        if time.time() - self._last_fetch_ts < 60:
            return self._cached_tasks

        url = f"{self.BASE_URL}/databases/{Config.NOTION_DATABASE_ID}/query"
        try:
            resp = self._session.post(url, headers=self.headers, json={"page_size": limit}, timeout=8)
            if resp.status_code != 200:
                logger.warning(f"Notion API respondió {resp.status_code}: {resp.text[:120]}")
                return []
            results = resp.json().get("results", [])
            tasks: List[Dict[str, str]] = []
            for idx, page in enumerate(results, start=1):
                props = page.get("properties", {})
                title = self._extract_title(props)
                tasks.append({
                    "id": f"N{idx}",
                    "title": f"[Notion] {title}",
                    "datetime": self._extract_date(props),
                    "status": "pendiente",
                })
            self._cached_tasks = tasks
            self._last_fetch_ts = time.time()
            return self._cached_tasks
        except Exception as exc:
            logger.error(f"Error consultando Notion: {exc}")
            return []

    def create_task(self, title: str, iso_date: str = "") -> Dict[str, Any]:
        """Crea una página/tarea directamente en la base de datos de Notion."""
        if not self.enabled:
            return {"ok": False, "error": "Notion no configurado en .env (usando agenda local)."}

        date_val = iso_date or datetime.now().strftime("%Y-%m-%d")
        url = f"{self.BASE_URL}/pages"
        payload = {
            "parent": {"database_id": Config.NOTION_DATABASE_ID},
            "properties": {
                "Name": {"title": [{"text": {"content": title.strip()}}]},
            },
        }
        try:
            resp = self._session.post(url, headers=self.headers, json=payload, timeout=8)
            if resp.status_code in (200, 201):
                self._last_fetch_ts = 0.0
                data = resp.json()
                logger.info(f"Tarea sincronizada en Notion: {title}")
                return {"ok": True, "url": data.get("url", ""), "date": date_val}
            return {"ok": False, "error": f"HTTP {resp.status_code}"}
        except Exception as exc:
            return {"ok": False, "error": str(exc)}

    @staticmethod
    def _extract_title(props: Dict[str, Any]) -> str:
        for prop in props.values():
            if prop.get("type") == "title":
                arr = prop.get("title", [])
                return "".join(t.get("plain_text", "") for t in arr) or "Sin título"
        return "Sin título"

    @staticmethod
    def _extract_date(props: Dict[str, Any]) -> str:
        for prop in props.values():
            if prop.get("type") == "date" and prop.get("date"):
                return prop["date"].get("start", "Hoy")
        return "Hoy"

"""
Ruta: modules/productivity_tools.py
Descripción: Gestión de agenda/calendario local, lectura de correos por IMAP
e inspección inteligente de archivos de código locales.
"""
import ast

from email.header import decode_header

import json
import logging

import webbrowser
from datetime import datetime
from pathlib import Path
from typing import Dict, Any, List
from config import Config, BASE_DIR, LOG_DIR

logger = logging.getLogger("ProductivityTools")
CALENDAR_FILE = LOG_DIR / "calendar_tasks.json"


class ProductivityTools:
    def __init__(self) -> None:
        self._cal_lock = threading.Lock()
        self._cached_events: List[Dict[str, Any]] | None = None
        self._ensure_calendar_file()

    def _ensure_calendar_file(self) -> None:
        with self._cal_lock:
            if not CALENDAR_FILE.exists():
                sample_events = [
                    {
                        "id": 1,
                        "title": "Revisar arquitectura de Argon 4 en VS Code",
                        "datetime": datetime.now().strftime("%Y-%m-%d 18:30"),
                        "status": "pendiente"
                    }
                ]
                CALENDAR_FILE.write_text(
                    json.dumps(sample_events, indent=2, ensure_ascii=False), encoding="utf-8"
                )

    # ---------------- CALENDARIO Y PENDIENTES ----------------
    def list_calendar_events(self) -> List[Dict[str, Any]]:
        """Devuelve la lista de eventos y pendientes activos usando caché en memoria."""
        with self._cal_lock:
            if self._cached_events is not None:
                return list(self._cached_events)
            try:
                data = json.loads(CALENDAR_FILE.read_text(encoding="utf-8"))
                self._cached_events = [item for item in data if item.get("status") == "pendiente"]
                return list(self._cached_events)
            except Exception as exc:
                logger.error(f"Error leyendo calendario: {exc}")
                return []

    def add_calendar_event(self, title: str, date_str: str = "") -> Dict[str, Any]:
        """Agrega un nuevo pendiente o cita a la agenda local."""
        with self._cal_lock:
            try:
                data = json.loads(CALENDAR_FILE.read_text(encoding="utf-8"))
                if not date_str:
                    date_str = datetime.now().strftime("%Y-%m-%d %H:%M")
                new_item = {
                    "id": len(data) + 1,
                    "title": title.strip(),
                    "datetime": date_str,
                    "status": "pendiente"
                }
                data.append(new_item)
                CALENDAR_FILE.write_text(
                    json.dumps(data, indent=2, ensure_ascii=False), encoding="utf-8"
                )
                self._cached_events = None
                logger.info(f"Pendiente agregado al calendario: {title}")
                return {"ok": True, "event": new_item}
            except Exception as exc:
                return {"ok": False, "error": str(exc)}

    def complete_calendar_event(self, event_id: int) -> Dict[str, Any]:
        """Marca un pendiente como completado por su ID."""
        with self._cal_lock:
            try:
                data = json.loads(CALENDAR_FILE.read_text(encoding="utf-8"))
                for item in data:
                    if item.get("id") == event_id:
                        item["status"] = "completado"
                        CALENDAR_FILE.write_text(
                            json.dumps(data, indent=2, ensure_ascii=False), encoding="utf-8"
                        )
                        self._cached_events = None
                        return {"ok": True, "title": item.get("title")}
                return {"ok": False, "error": f"No se encontró el pendiente #{event_id}."}
            except Exception as exc:
                return {"ok": False, "error": str(exc)}

    # ---------------- REVISIÓN DE CORREOS (IMAP / WEB) ----------------
    @staticmethod
    def check_emails(limit: int = 3) -> Dict[str, Any]:
        """
        Consulta los últimos correos vía IMAP si las credenciales están en .env.
        Si no están configuradas, abre la bandeja de entrada en el navegador.
        """
        if not Config.EMAIL_USER or not Config.EMAIL_APP_PASSWORD:
            webbrowser.open("https://mail.google.com")
            return {
                "ok": True,
                "mode": "browser",
                "summary": (
                    "He abierto tu bandeja de correo en el navegador. "
                    "Si deseas que lea tus correos en voz alta sin abrir el navegador, "
                    "coloca tu EMAIL_USER y EMAIL_APP_PASSWORD en el archivo .env."
                )
            }

        try:
            mail = imaplib.IMAP4_SSL(Config.EMAIL_IMAP_HOST, timeout=10)
            mail.login(Config.EMAIL_USER, Config.EMAIL_APP_PASSWORD)
            mail.select("inbox")

            status, messages = mail.search(None, "UNSEEN")
            mail_ids = messages[0].split()
            if not mail_ids:
                mail.logout()
                return {"ok": True, "mode": "imap", "summary": "No tienes correos nuevos sin leer."}

            recent_ids = mail_ids[-limit:]
            summaries = []
            for m_id in reversed(recent_ids):
                _, msg_data = mail.fetch(m_id, "(RFC822)")
                for response_part in msg_data:
                    if isinstance(response_part, tuple):
                        msg = email.message_from_bytes(response_part[1])
                        subject, encoding = decode_header(msg.get("Subject", "Sin asunto"))[0]
                        if isinstance(subject, bytes):
                            subject = subject.decode(encoding or "utf-8", errors="ignore")
                        sender = msg.get("From", "Remitente desconocido")
                        summaries.append(f"De {sender}: {subject}")

            mail.logout()
            joined = " | ".join(summaries)
            return {
                "ok": True,
                "mode": "imap",
                "summary": f"Tienes {len(mail_ids)} correos sin leer. Los más recientes son: {joined}"
            }
        except Exception as exc:
            logger.error(f"Error IMAP: {exc}")
            return {"ok": False, "error": f"No pude conectar al servidor de correo: {exc}"}

    # ---------------- AUDITOR Y ANALIZADOR DE CÓDIGO ----------------
    @staticmethod
    def analyze_code_file(relative_path: str) -> Dict[str, Any]:
        """
        Inspecciona un archivo de código del proyecto, verifica su sintaxis AST
        y extrae su contenido bloqueando el acceso a archivos sensibles (.env).
        """
        clean_name = relative_path.strip().replace('"', "").replace("'", "")
        if ".env" in clean_name.lower():
            return {"ok": False, "error": "Acceso denegado: Por seguridad no se permite inspeccionar archivos .env."}

        base_resolved = BASE_DIR.resolve()
        target = (BASE_DIR / clean_name).resolve()

        if not str(target).startswith(str(base_resolved)) or not target.exists() or not target.is_file():
            matches = [
                m for m in BASE_DIR.rglob(f"*{Path(clean_name).name}*")
                if ".venv" not in str(m) and ".env" not in m.name.lower() and m.is_file()
            ]
            if matches:
                target = matches[0].resolve()
            else:
                return {"ok": False, "error": f"No encontré el archivo '{clean_name}' en el proyecto."}

        try:
            content = target.read_text(encoding="utf-8", errors="ignore")
            lines = content.splitlines()
            syntax_status = "Sintaxis válida"

            if target.suffix == ".py":
                try:
                    tree = ast.parse(content)
                    funcs = [n.name for n in ast.walk(tree) if isinstance(n, ast.FunctionDef)]
                    classes = [n.name for n in ast.walk(tree) if isinstance(n, ast.ClassDef)]
                    syntax_status = (
                        f"Sintaxis Python correcta. Contiene {len(classes)} clases "
                        f"({', '.join(classes[:4])}) y {len(funcs)} funciones."
                    )
                except SyntaxError as syn_err:
                    syntax_status = f"Error de sintaxis en línea {syn_err.lineno}: {syn_err.msg}"

            return {
                "ok": True,
                "filepath": str(target.relative_to(base_resolved)),
                "lines": len(lines),
                "syntax": syntax_status,
                "snippet": "\n".join(lines[:90])
            }
        except Exception as exc:
            return {"ok": False, "error": str(exc)}

    @staticmethod

        """Obtiene métricas reales de CPU, RAM, disco y batería sin bloquear el hilo."""
        try:
            cpu_usage = psutil.cpu_percent(interval=None)
            memory = psutil.virtual_memory()
            disk = psutil.disk_usage(str(BASE_DIR))
            battery = psutil.sensors_battery()

            status = {
                "cpu_percent": cpu_usage,
                "ram_percent": memory.percent,
                "ram_available_gb": round(memory.available / (1024 ** 3), 2),
                "disk_free_gb": round(disk.free / (1024 ** 3), 2),
                "battery_percent": int(battery.percent) if battery else 100,
                "power_plugged": battery.power_plugged if battery else True,
            }
            logger.debug(f"Telemetría recolectada: {status}")
            return {"ok": True, "data": status}
        except Exception as exc:
            logger.error(f"Error al leer telemetría del sistema: {exc}")
            return {"ok": False, "error": str(exc)}

        except Exception as exc:

    @staticmethod
    def open_application(app_name: str) -> Dict[str, Any]:
        """Abre aplicaciones estándar del sistema según la plataforma actual."""
        system_os = platform.system().lower()
        app_key = app_name.lower().strip()

        # Mapa seguro de binarios comunes
        app_map = {
            "windows": {
                "calculadora": ["calc.exe"],
                "bloc de notas": ["notepad.exe"],
                "explorador": ["explorer.exe"],
                "terminal": ["cmd.exe", "/c", "start", "cmd.exe"],
            },
            "linux": {
                "calculadora": ["gnome-calculator"],
                "bloc de notas": ["gedit"],
                "explorador": ["xdg-open", str(Path.home())],
                "terminal": ["x-terminal-emulator"],
            },
            "darwin": {
                "calculadora": ["open", "-a", "Calculator"],
                "bloc de notas": ["open", "-a", "TextEdit"],
                "explorador": ["open", str(Path.home())],
                "terminal": ["open", "-a", "Terminal"],
            },
        }

        commands = app_map.get(system_os, {}).get(app_key)
        if not commands:
            return {
                "ok": False,
                "error": f"Aplicación '{app_name}' no registrada en la lista blanca para {system_os}."
            }

        try:
            subprocess.Popen(
                commands,
                stdout=subprocess.DEVNULL,
                stderr=subprocess.DEVNULL,
                shell=(system_os == "windows" and app_key == "terminal")
            )
            logger.info(f"Aplicación ejecutada: {app_name}")
            return {"ok": True, "app": app_name}
        except FileNotFoundError:
            return {"ok": False, "error": f"No se encontró el binario para '{app_name}'."}
        except Exception as exc:
            logger.error(f"Error lanzando aplicación {app_name}: {exc}")
            return {"ok": False, "error": str(exc)}

    @staticmethod
    def open_external_protocol(uri: str) -> Dict[str, Any]:
        """
        Dispara enlaces de protocolo local (ej. sip:usuario@dominio, whatsapp://send?phone=...).
        """
        try:
            webbrowser.open(uri)
            logger.info(f"Enlace de protocolo externo invocado: {uri}")
            return {"ok": True, "uri": uri}
        except Exception as exc:
            logger.error(f"Error al abrir URI {uri}: {exc}")
            return {"ok": False, "error": str(exc)}

    # ---------------- DESPACHO PARALELO ----------------
    def execute_parallel_calls(self, normalized_calls: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
        """Ejecuta múltiples herramientas simultáneamente en hilos paralelos."""
        results: List[Dict[str, Any]] = []
        futures = {}

        for call in normalized_calls:
            fn_name = call.get("name", "")
            args = call.get("args", {}) or {}

            fn = self.registry.get(fn_name)
            if not fn:
                results.append({"tool": fn_name, "result": {"ok": False, "error": "Herramienta no registrada."}})
                continue

            self.core._set_tool_telemetry(fn_name.upper(), f"Ejecutando {fn_name}...")
            future = self.executor.submit(fn, **args)
            futures[future] = fn_name

        for future in as_completed(futures):
            fn_name = futures[future]
            try:
                res = future.result(timeout=35)
                results.append({"tool": fn_name, "result": res})
            except Exception as exc:
                logger.error(f"Error en herramienta {fn_name}: {exc}")
                results.append({"tool": fn_name, "result": {"ok": False, "error": str(exc)}})

        return results

    def route_and_execute(self, user_text: str) -> str | None:
        """
        1. Si GEMINI_API_KEY está activa, usa Function Calling nativo de Google Gemini.
        2. Si no, usa Function Calling de Ollama local.
        """
        # 1. Google Gemini Native Tool Calling
        if self.core.gemini_online:
            try:
                url = (
                    f"https://generativelanguage.googleapis.com/v1beta/models/"
                    f"{Config.GEMINI_MODEL}:generateContent?key={Config.GEMINI_API_KEY}"
                )
                payload = {
                    "systemInstruction": {"parts": [{"text": self.core.system_prompt}]},
                    "contents": [{"role": "user", "parts": [{"text": user_text}]}],
                    "tools": [{"functionDeclarations": self.function_declarations}],
                }
                resp = self.core.http.post(url, json=payload, timeout=20)
                if resp.status_code == 200:
                    candidates = resp.json().get("candidates", [])
                    if candidates:
                        parts = candidates[0].get("content", {}).get("parts", [])
                        gemini_calls = [
                            {"name": p["functionCall"]["name"], "args": p["functionCall"].get("args", {})}
                            for p in parts if "functionCall" in p
                        ]
                        if gemini_calls:
                            tool_outputs = self.execute_parallel_calls(gemini_calls)
                            context_str = json.dumps(tool_outputs, ensure_ascii=False)
                            return self.core.ask_brain(
                                f"Confirma al usuario de forma ejecutiva el resultado de su solicitud: '{user_text}'",
                                extra_context=f"[RESULTADOS DE HERRAMIENTAS EJECUTADAS]: {context_str}"
                            )
            except Exception as exc:
                logger.debug(f"Gemini Tool Router fallback: {exc}")

        # 2. Ollama Local Tool Calling
        if not self.core.ollama_online:
            return None

        try:
            payload = {
                "model": self.core.active_model,
                "messages": [
                    {"role": "system", "content": self.core.system_prompt},
                    {"role": "user", "content": user_text}
                ],
                "tools": self.ollama_tools,
                "stream": False
            }
            resp = self.core.http.post(
                f"{Config.OLLAMA_HOST}/api/chat",
                json=payload,
                timeout=Config.OLLAMA_TIMEOUT
            )
            if resp.status_code != 200:
                return None

            raw_calls = resp.json().get("message", {}).get("tool_calls", [])
            if not raw_calls:
                return None

            ollama_calls = []
            for c in raw_calls:
                fn_info = c.get("function", {})
                raw_args = fn_info.get("arguments", {})
                args = json.loads(raw_args) if isinstance(raw_args, str) else (raw_args or {})
                ollama_calls.append({"name": fn_info.get("name", ""), "args": args})

            tool_outputs = self.execute_parallel_calls(ollama_calls)
            context_str = json.dumps(tool_outputs, ensure_ascii=False)
            return self.core.ask_brain(
                f"Confirma al usuario de forma ejecutiva el resultado de su solicitud: '{user_text}'",
                extra_context=f"[RESULTADOS DE HERRAMIENTAS EJECUTADAS]: {context_str}"
            )
        except Exception as exc:
            logger.debug(f"Ollama Tool Router fallback: {exc}")
            return None

        self.tts_available = False
        self._speak_lock = threading.Lock()
        self._listen_lock = threading.Lock()
        self._abort_speech = threading.Event()

        # Configuración de timbre grave futurista (estilo J.A.R.V.I.S.)
        self.voice_name: str = getattr(Config, "ASSISTANT_VOICE", "es-ES-AlvaroNeural")
        if self.voice_name == "es-PE-AlexNeural":
            self.voice_name = "es-ES-AlvaroNeural"
        self.voice_pitch: str = getattr(Config, "VOICE_PITCH", "-14Hz")
        self.voice_rate: str = getattr(Config, "VOICE_RATE_STR", "+6%")

        try:
            pygame.mixer.init(frequency=24000, size=-16, channels=2, buffer=1024)
            self.neural_audio_ready = True
        except Exception as exc:
            logger.warning(f"No se pudo iniciar pygame.mixer ({exc}).")
            self.neural_audio_ready = False

        self._init_fallback_tts()
        self._check_microphone()

    def set_voice_profile(self, voice_name: str, pitch: str = "-14Hz", rate: str = "+6%") -> None:
        """Permite ajustar la profundidad de la voz en caliente desde el HUD."""
        self.voice_name = voice_name
        self.voice_pitch = pitch
        self.voice_rate = rate
        logger.info(f"Perfil de voz actualizado: {voice_name} | Pitch: {pitch} | Rate: {rate}")

    def stop_speaking(self) -> None:
        """Interrupción instantánea (Barge-In): corta cualquier locución activa en <10ms."""
        self._abort_speech.set()
        if self.neural_audio_ready:
            try:
                if pygame.mixer.music.get_busy():
                    pygame.mixer.music.stop()
            except Exception:
                pass
        if self.tts_available:
            try:
                self.engine.stop()
            except Exception:
                pass
        self._notify_state("IDLE")

    def _notify_state(self, state: str) -> None:
        if self.on_state_change:
            try:
                self.on_state_change(state)
            except Exception:
                pass

    def _init_fallback_tts(self) -> None:
        try:
            self.engine = pyttsx3.init()
            self.engine.setProperty("rate", 165)
            self.engine.setProperty("volume", 1.0)
            self.tts_available = True
        except Exception:
            self.tts_available = False

    def _check_microphone(self) -> None:
        try:
            mics = sr.Microphone.list_microphone_names()
            self.mic_available = bool(mics)
        except Exception:
            self.mic_available = False

    async def _synthesize_edge_tts(self, text: str, output_path: str) -> None:
        communicate = edge_tts.Communicate(
            text=text,
            voice=self.voice_name,
            pitch=self.voice_pitch,
            rate=self.voice_rate
        )
        await communicate.save(output_path)

    def speak(self, text: str) -> None:
        """Sintetiza voz neuronal con frecuencia grave (-14Hz) e interrupción Barge-In."""
        if not text:
            return
        clean_text = text.replace("*", "").replace("#", "").strip()
        if not clean_text:
            return

        # Si ya estaba hablando, interrumpir la locución anterior para priorizar la nueva
        self.stop_speaking()

        with self._speak_lock:
            self._abort_speech.clear()
            print(f"\n[{Config.AGENT_NAME}]: {clean_text}")
            self._notify_state("SPEAKING")

            if self.neural_audio_ready:
                temp_file = None
                try:
                    with tempfile.NamedTemporaryFile(delete=False, suffix=".mp3") as fp:
                        temp_file = fp.name

                    asyncio.run(self._synthesize_edge_tts(clean_text, temp_file))

                    if self._abort_speech.is_set():
                        if temp_file and os.path.exists(temp_file):
                            os.remove(temp_file)
                        self._notify_state("IDLE")
                        return

                    pygame.mixer.music.load(temp_file)
                    pygame.mixer.music.play()
                    while pygame.mixer.music.get_busy():
                        if self._abort_speech.is_set():
                            pygame.mixer.music.stop()
                            break
                        pygame.time.Clock().tick(30)
                    pygame.mixer.music.unload()
                    if temp_file and os.path.exists(temp_file):
                        os.remove(temp_file)
                    self._notify_state("IDLE")
                    return
                except Exception as exc:
                    logger.warning(f"Fallo en síntesis neuronal ({exc}). Usando respaldo.")
                    if temp_file and os.path.exists(temp_file):
                        try:
                            os.remove(temp_file)
                        except OSError:
                            pass

            if self.tts_available and not self._abort_speech.is_set():
                try:
                    self.engine.say(clean_text)
                    self.engine.runAndWait()
                except Exception as exc:
                    logger.error(f"Error TTS offline: {exc}")

            self._notify_state("IDLE")

    def listen(self, timeout: int = 6, phrase_time_limit: int = 15) -> Optional[str]:
        if not self.mic_available:
            return None
        # Cortar la voz del agente inmediatamente al iniciar la escucha (Barge-In)
        self.stop_speaking()

        if not self._listen_lock.acquire(blocking=False):
            return None
        try:
            self._notify_state("LISTENING")
            with sr.Microphone() as source:
                self.recognizer.adjust_for_ambient_noise(source, duration=0.3)
                audio = self.recognizer.listen(
                    source, timeout=timeout, phrase_time_limit=phrase_time_limit
                )
            self._notify_state("THINKING")
            query = self.recognizer.recognize_google(audio, language=Config.LANGUAGE_CODE)
            logger.info(f"Voz reconocida: '{query}'")
            self._notify_state("IDLE")
            return query.strip()
        except (sr.WaitTimeoutError, sr.UnknownValueError):
            self._notify_state("IDLE")
            return None
        except Exception as exc:
            logger.error(f"Error en captura de voz: {exc}")
            self._notify_state("IDLE")
            return None
        finally:
            self._listen_lock.release()

"""
Ruta: modules/voip_caller.py
Descripción: Enrutador de llamadas VoIP/SIP y WebRTC de costo cero mediante Linphone CLI,
protocolos nativos del sistema operativo (sip:/tel:) y salas WebRTC abiertas.
"""
import re
import uuid
import shutil
import subprocess
import webbrowser
import logging
from typing import Dict, Any, Optional
from config import Config

logger = logging.getLogger("VoIPCaller")


class VoIPCaller:
    def __init__(self) -> None:
        self.softphone_bin = Config.SIP_SOFTPHONE_BIN
        self.cli_available = shutil.which(self.softphone_bin) is not None
        if self.cli_available:
            logger.info(f"Binario VoIP CLI detectado: {self.softphone_bin}")
        else:
            logger.info(
                "Linphone CLI no detectado en PATH. Se usará despacho por protocolo SIP/TEL nativo y WebRTC."
            )

    @staticmethod
    def format_target(raw_target: str) -> Optional[str]:
        """
        Determina si el destino es una dirección SIP (usuario@dominio) o un número telefónico.
        """
        cleaned = raw_target.strip().replace(" ", "")
        if not cleaned:
            return None

        # Si es una cuenta SIP (ej. usuario@sip.linphone.org)
        if "@" in cleaned:
            return cleaned if cleaned.startswith("sip:") else f"sip:{cleaned}"

        # Si es un número telefónico
        digits = re.sub(r"[^\d+]", "", cleaned)
        if len(digits) >= 6:
            if not digits.startswith("+") and len(digits) == 9:
                digits = f"{Config.DEFAULT_COUNTRY_CODE}{digits}"
            return digits

        return None

    def start_call(self, target: str, mode: str = "auto") -> Dict[str, Any]:
        """
        Inicia una llamada gratuita usando:
        - 'cli': Linphone CLI local (si está disponible)
        - 'protocol': URI sip: o tel: (abre softphone de escritorio o Enlace Móvil de Windows)
        - 'webrtc': Crea una sala de voz WebRTC instantánea sin registro
        """
        if mode == "webrtc":
            return self.start_webrtc_room()

        formatted = self.format_target(target)
        if not formatted:
            return {
                "ok": False,
                "error": f"Destino de llamada inválido ('{target}'). Indica un número o cuenta SIP."
            }

        # 1. Intentar por Linphone CLI si está instalado en el sistema
        if self.cli_available and mode in ("auto", "cli"):
            try:
                subprocess.run(
                    [self.softphone_bin, "init"],
                    stdout=subprocess.DEVNULL,
                    stderr=subprocess.DEVNULL,
                    timeout=3
                )
                res = subprocess.run(
                    [self.softphone_bin, "generic", f"call {formatted}"],
                    capture_output=True,
                    text=True,
                    timeout=5
                )
                if res.returncode == 0:
                    logger.info(f"Llamada SIP originada por CLI hacia {formatted}")
                    return {"ok": True, "method": "Linphone CLI", "target": formatted}
            except Exception as exc:
                logger.warning(f"Fallo en llamada CLI ({exc}). Conmutando a protocolo nativo...")

        # 2. Enrutamiento por protocolo nativo del SO (sip: o tel:)
        try:
            uri = formatted if formatted.startswith("sip:") else f"tel:{formatted}"
            webbrowser.open(uri)
            logger.info(f"Protocolo de llamada invocado: {uri}")
            return {"ok": True, "method": "Protocolo Nativo (SIP/TEL)", "target": uri}
        except Exception as exc:
            logger.error(f"Error al disparar protocolo de llamada: {exc}")
            return {"ok": False, "error": str(exc)}

    def hangup_call(self) -> Dict[str, Any]:
        """Corta una llamada activa en Linphone CLI."""
        if not self.cli_available:
            return {
                "ok": False,
                "error": "El corte por comando requiere Linphone CLI activo; cierra desde la ventana del softphone."
            }
        try:
            subprocess.run(
                [self.softphone_bin, "generic", "terminate"],
                capture_output=True,
                text=True,
                timeout=4
            )
            logger.info("Llamada finalizada por comando CLI.")
            return {"ok": True}
        except Exception as exc:
            return {"ok": False, "error": str(exc)}

    @staticmethod
    def start_webrtc_room() -> Dict[str, Any]:
        """
        Genera y abre un canal de llamada de voz/video WebRTC gratuito sin cuenta ni tarjeta.
        """
        try:
            room_id = f"Argon4-Voice-{uuid.uuid4().hex[:8]}"
            room_url = f"https://meet.jit.si/{room_id}#config.startWithVideoMuted=true"
            webbrowser.open(room_url)
            logger.info(f"Sala de voz WebRTC abierta: {room_url}")
            return {"ok": True, "method": "WebRTC Directo", "url": room_url}
        except Exception as exc:
            logger.error(f"Error al crear sala WebRTC: {exc}")
            return {"ok": False, "error": str(exc)}

"""
Ruta: modules/whatsapp_bridge.py
Descripción: Automatización local de WhatsApp Web sin APIs de pago, con normalización
de números internacionales y respaldo por protocolo URI.
"""
import os
import re
import time
import urllib.parse
import webbrowser
import logging
from typing import Dict, Any, Optional
import pyautogui
import pywhatkit
import requests
from config import Config

logger = logging.getLogger("WhatsAppBridge")


class WhatsAppBridge:
    def __init__(self) -> None:
        # Agenda rápida local: puedes agregar aquí tus contactos frecuentes
        self.contacts: Dict[str, str] = {
            "prueba": os.getenv("TEST_PHONE", ""),
        }

    @staticmethod
    def normalize_phone(raw_input: str) -> Optional[str]:
        """
        Limpia el texto dictado o escrito y lo convierte a formato E.164 (ej. +51999888777).
        """
        digits = re.sub(r"[^\d+]", "", raw_input.strip())
        if not digits:
            return None

        if digits.startswith("+"):
            return digits if len(digits) >= 10 else None

        # Si ya incluye el código de país sin el '+' (ej. 51999888777)
        clean_cc = Config.DEFAULT_COUNTRY_CODE.replace("+", "")
        if digits.startswith(clean_cc) and len(digits) >= 11:
            return f"+{digits}"

        # Número celular local (ej. 9 dígitos en Perú)
        if len(digits) >= 9:
            return f"{Config.DEFAULT_COUNTRY_CODE}{digits}"

        return None

    def send_message(self, target_phone: str, message: str, wait_time: int = 15) -> Dict[str, Any]:
        """
        Envía un mensaje de WhatsApp intentando primero el puente REST local (Baileys)
        y conmutando a automatización web (pywhatkit / PyAutoGUI) si no está activo.
        """
        phone = self.normalize_phone(target_phone)
        if not phone:
            return {
                "ok": False,
                "error": f"Número inválido ('{target_phone}'). Dicta o escribe al menos 9 dígitos."
            }

        if not message or not message.strip():
            return {"ok": False, "error": "El mensaje está vacío."}

        logger.info(f"Iniciando envío de WhatsApp a {phone}...")

        # 1. Intento vía puente REST local (Baileys)
        try:
            bridge_url = f"{Config.WHATSAPP_BRIDGE_URL}/send"
            payload = {
                "session": Config.WHATSAPP_SESSION,
                "phone": phone,
                "message": message.strip()
            }
            resp = requests.post(bridge_url, json=payload, timeout=4)
            if resp.status_code == 200:
                logger.info(f"WhatsApp enviado con éxito vía puente REST local a {phone}.")
                return {"ok": True, "phone": phone, "message": message.strip()}
            logger.warning(f"Puente REST respondió con código {resp.status_code}. Continuando con navegador...")
        except requests.RequestException as exc:
            logger.info(f"Puente REST local de Baileys no disponible ({exc}). Continuando con flujo de navegador...")

        # 2. Flujo de navegador (pywhatkit + respaldo por URL)
        try:
            # Método principal: abre WhatsApp Web, espera carga, escribe, envía y cierra pestaña
            pywhatkit.sendwhatmsg_instantly(
                phone_no=phone,
                message=message.strip(),
                wait_time=wait_time,
                tab_close=True,
                close_time=3
            )
            logger.info(f"WhatsApp enviado con éxito a {phone}.")
            return {"ok": True, "phone": phone, "message": message.strip()}

        except Exception as exc:
            logger.warning(f"Fallo en pywhatkit ({exc}). Intentando respaldo por URL directa...")
            return self._fallback_web_send(phone, message.strip(), wait_time)

    @staticmethod
    def _fallback_web_send(phone: str, message: str, wait_time: int) -> Dict[str, Any]:
        """Respaldo directo abriendo la URL de WhatsApp Web y presionando Enter con PyAutoGUI."""
        try:
            encoded_msg = urllib.parse.quote(message)
            url = f"https://web.whatsapp.com/send?phone={phone.replace('+', '')}&text={encoded_msg}"
            webbrowser.open(url)
            time.sleep(wait_time)
            pyautogui.press("enter")
            logger.info(f"WhatsApp enviado por vía de respaldo a {phone}.")
            return {"ok": True, "phone": phone, "message": message}
        except Exception as exc:
            logger.error(f"Error crítico enviando WhatsApp: {exc}")
            return {"ok": False, "error": str(exc)}
